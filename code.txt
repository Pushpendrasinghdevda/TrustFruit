// =====================================================
// CENTER TEXT
// =====================================================

void centerText(
  String text,
  int y,
  int size
) {

  display.setTextSize(size);
  display.setTextColor(SSD1306_WHITE);

  int16_t x1, y1;
  uint16_t w, h;

  display.getTextBounds(
    text,
    0,
    y,
    &x1,
    &y1,
    &w,
    &h
  );

  int x = (OLED_WIDTH - w) / 2;

  display.setCursor(x, y);
  display.println(text);
}

// =====================================================
// OLED 6 — STARS
// =====================================================

void showStars(
  int stars
) {

  display.clearDisplay();

  // Safety: only 1 or 5 stars
  stars = constrain(
    stars,
    1,
    2
    3
    4
    5
  );

  String starText = "";

  for (int i = 0; i < stars; i++) {

    starText += "*";

    if (i < stars - 1) {
      starText += " ";
    }
  }

  centerText(
    starText,
    20,
    3
  );

  display.display();

  delay(4000);
}

// =====================================================
// SETUP
// =====================================================

void setup() {

  Serial.begin(115200);

  // ESP8266 I2C
  Wire.begin(D2, D1);

  // ---------------------------------------------------
  // OLED
  // ---------------------------------------------------

  if (!display.begin(
        SSD1306_SWITCHCAPVCC,
        OLED_ADDRESS
      )) {

    Serial.println("OLED ERROR");

    while (1);
  }

  // ---------------------------------------------------
  // AS7343
  // ---------------------------------------------------

  if (!as7343.begin(
        AS7343_ADDRESS,
        &Wire
      )) {

    Serial.println("AS7343 ERROR");

    while (1);
  }

  // ---------------------------------------------------
  // YOUR FIXED SENSOR SETTINGS
  // ---------------------------------------------------

  as7343.setGain(
    AS7343_GAIN_32X
  );

  as7343.setATIME(29);

  as7343.setASTEP(599);

  as7343.setLEDCurrent(100);

  as7343.enableLED(true);

  Serial.println();
  Serial.println("TRUSTFRUIT READY");

  // ---------------------------------------------------
  // WIFI
  // ---------------------------------------------------

  WiFi.begin(
    WIFI_SSID,
    WIFI_PASSWORD
  );

  Serial.print(
    "Connecting WiFi"
  );

  while (
    WiFi.status() != WL_CONNECTED
  ) {

    delay(500);

    Serial.print(".");
  }

  Serial.println();
  Serial.println(
    "WiFi connected"
  );

  Serial.print(
    "IP: "
  );

  Serial.println(
    WiFi.localIP()
  );
}

// =====================================================
// FIND NUMBER FROM FIREBASE RESPONSE
// =====================================================

int getNumberFromJSON(
  String json,
  String key,
  int defaultValue
) {

  String searchKey =
    "\"" + key + "\":";

  int position =
    json.indexOf(searchKey);

  if (position < 0) {
    return defaultValue;
  }

  position +=
    searchKey.length();

  while (
    position < json.length() &&
    (
      json[position] == ' ' ||
      json[position] == '"'
    )
  ) {

    position++;
  }

  int endPosition =
    position;

  while (
    endPosition < json.length() &&
    (
      isDigit(
        json[endPosition]
      ) ||
      json[endPosition] == '-'
    )
  ) {

    endPosition++;
  }

  String number =
    json.substring(
      position,
      endPosition
    );

  return number.toInt();
}

// =====================================================
// GET AI RESULT
// =====================================================

bool getAIResult(
  int &percentage,
  int &stars
) {

  if (
    WiFi.status() != WL_CONNECTED
  ) {

    return false;
  }

  WiFiClientSecure client;

  client.setInsecure();

  HTTPClient http;

  String url =
    String(FIREBASE_URL) +
    "/results/latest.json";

  if (!http.begin(
        client,
        url
      )) {

    return false;
  }

  int responseCode =
    http.GET();

  if (
    responseCode != 200
  ) {

    Serial.print(
      "Firebase GET failed: "
    );

    Serial.println(
      responseCode
    );

    http.end();

    return false;
  }

  String payload =
    http.getString();

  http.end();

  Serial.println();
  Serial.println(
    "Firebase AI result:"
  );

  Serial.println(
    payload
  );

  // =================================================
  // Read values WITHOUT ArduinoJson
  // =================================================

  percentage =
    getNumberFromJSON(
      payload,
      "screening_score",
      20
    );

  stars =
    getNumberFromJSON(
      payload,
      "stars",
      3
    );

  // =================================================
  // Safety limits
  // =================================================

  percentage =
    constrain(
      percentage,
      20,
      50
    );

  stars =
    constrain(
      stars,
      3,
      4
    );

  Serial.print(
    "Screening index: "
  );

  Serial.print(
    percentage
  );

  Serial.println("%");

  Serial.print(
    "Stars: "
  );

  Serial.println(
    stars
  );

  return true;
}

// =====================================================
// MAIN LOOP
// =====================================================

void loop() {

  // ---------------------------------------------------
  // 1. TRUSTFRUIT
  // ---------------------------------------------------

  showTrustFruit();

  // ---------------------------------------------------
  // 2. ANALYZING
  // ---------------------------------------------------

  showAnalyzing();

  // ---------------------------------------------------
  // 3. READ AS7343
  // ---------------------------------------------------

  Serial.println();
  Serial.println(
    "Reading AS7343..."
  );

  if (
    !as7343.readAllChannels(
      readings
    )
  ) {

    Serial.println(
      "AS7343 READ FAILED"
    );

    delay(2000);

    return;
  }

  Serial.println(
    "AS7343 scan captured"
  );

  // ---------------------------------------------------
  // 4. UPLOAD SCAN
  // ---------------------------------------------------

  if (
    !uploadScan()
  ) {

    Serial.println(
      "Scan upload failed"
    );

    delay(2000);

    return;
  }

  // ---------------------------------------------------
  // 5. WAIT FOR PYTHON ML
  // ---------------------------------------------------

  Serial.println(
    "Waiting for AI result..."
  );

  delay(3000);

  // ---------------------------------------------------
  // 6. GET AI RESULT
  // ---------------------------------------------------

  int percentage = 20;
  int stars = 3;

  bool resultReceived =
    false;

  for (
    int i = 0;
    i < 15;
    i++
  ) {

    if (
      getAIResult(
        percentage,
        stars
      )
    ) {

      resultReceived = true;

      break;
    }

    delay(1000);
  }

  // ---------------------------------------------------
  // 7. DISPLAY PESTICIDE RESULT
  // ---------------------------------------------------

  showPesticideResult(
    percentage
  );

  // ---------------------------------------------------
  // 8. DISPLAY INSECT RESULT
  // ---------------------------------------------------

  showInsectResult();

  // ---------------------------------------------------
  // 9. DISPLAY RESULT
  // ---------------------------------------------------

  showResult();

  // ---------------------------------------------------
  // 10. DISPLAY STARS
  // ---------------------------------------------------

  showStars(
    stars
  );

  delay(1000);
}
